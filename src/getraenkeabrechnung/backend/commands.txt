create table user (
    UserID varchar(40) primary key,
    UserName varchar(40),
    Adults int(2),
    Children int(2),
    Image varchar(64)
);
create table drinks (
    DrinkID varchar(40) primary key,
    DrinkName varchar(40),
    Price float(4),
    Image varchar(64)
);
create table entry (
    ID varchar(40) primary key,
    UserID varchar(40),
    DrinkID varchar(40),
    OrderTime TIMESTAMP
);

SELECT u.UserName, sum(d.Price) AS Gesamtbetrag
FROM 
    entry AS e, 
    user AS u, 
    drinks AS d
WHERE 
    e.UserID = u.UserID AND
    e.DrinkID = d.DrinkID
GROUP BY u.UserID
--HAVING u.UserName = "Jung"
ORDER BY 2 DESC;


SELECT u.UserName, d.DrinkName, d.Price, e.OrderTime
FROM 
    entry AS e, 
    user AS u, 
    drinks AS d
WHERE 
    e.UserID = u.UserID AND
    e.DrinkID = d.DrinkID AND
    u.UserName = "Jung";

SELECT 
    u.UserName AS Name, 
    sum(d.Price) AS Gesamtbetrag
FROM 
    entry AS e, 
    user AS u, 
    drinks AS d
WHERE 
    e.UserID = u.UserID AND
    e.DrinkID = d.DrinkID
GROUP BY u.UserID
--HAVING u.UserName = "Jung"
ORDER BY 2 DESC;

SELECT 
    UserName, 
    DrinkName, 
    Price,
    count(d.DrinkID) as Anzahl,
    d.AdultTag,
    OrderTime
FROM
    entry AS e, 
    user AS u, 
    drinks AS d
WHERE 
    e.UserID = u.UserID AND
    e.DrinkID = d.DrinkID
